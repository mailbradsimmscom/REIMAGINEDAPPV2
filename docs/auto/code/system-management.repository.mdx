import { getSupabaseClient } from './supabaseClient.js';
import { logger } from '../utils/logger.js';

/**
 * System Management Repository
 *
 * All database operations for system management feature.
 * Follows cursor rules: ALL DB/storage/network I/O goes here.
 */

/**
 * Get distinct manufacturers from systems table
 * @returns {Promise<Array<string>>} Array of manufacturer names
 */
export async function getManufacturers() {
  const supabase = await getSupabaseClient();
  const requestLogger = logger.createRequestLogger();

  try {
    const { data, error } = await supabase
      .from('systems')
      .select('manufacturer_norm')
      .not('manufacturer_norm', 'is', null)
      .order('manufacturer_norm');

    if (error) throw error;

    // Get unique manufacturers
    const manufacturers = [...new Set(data.map(row => row.manufacturer_norm))];
    return manufacturers.sort();

  } catch (error) {
    requestLogger.error('Repository error fetching manufacturers', { error: error.message });
    throw error;
  }
}

/**
 * Get models for a specific manufacturer
 * @param {string} manufacturer - Manufacturer name
 * @returns {Promise<Array<string>>} Array of model names
 */
export async function getModelsByManufacturer(manufacturer) {
  const supabase = await getSupabaseClient();
  const requestLogger = logger.createRequestLogger();

  try {
    const { data, error } = await supabase
      .from('systems')
      .select('model_norm')
      .eq('manufacturer_norm', manufacturer)
      .not('model_norm', 'is', null)
      .order('model_norm');

    if (error) throw error;

    const models = [...new Set(data.map(row => row.model_norm))];
    return models.sort();

  } catch (error) {
    requestLogger.error('Repository error fetching models', {
      error: error.message,
      manufacturer
    });
    throw error;
  }
}

/**
 * Archive an instance (soft delete)
 * @param {string} instanceUid - Instance UID to archive
 * @returns {Promise<void>}
 */
export async function archiveInstance(instanceUid) {
  const supabase = await getSupabaseClient();
  const requestLogger = logger.createRequestLogger();

  try {
    // First, get the full instance record
    const { data: instance, error: fetchError } = await supabase
      .from('instances')
      .select('*')
      .eq('instance_uid', instanceUid)
      .single();

    if (fetchError) throw fetchError;
    if (!instance) throw new Error('Instance not found');

    // Insert into archive table (only essential fields + original_data JSONB)
    const { error: archiveError } = await supabase
      .from('instances_archived')
      .insert([{
        instance_uid: instance.instance_uid,
        asset_uid: instance.asset_uid,
        serial_number: instance.serial_number,
        location: instance.location,
        instance_index: instance.instance_index,
        archived_at: new Date().toISOString(),
        archived_by: 'system',
        original_data: instance
      }]);

    if (archiveError) throw archiveError;

    // Delete from instances table
    const { error: deleteError } = await supabase
      .from('instances')
      .delete()
      .eq('instance_uid', instanceUid);

    if (deleteError) throw deleteError;

  } catch (error) {
    requestLogger.error('Repository error archiving instance', {
      error: error.message,
      instanceUid
    });
    throw error;
  }
}

/**
 * Get next instance_index for a system
 * @param {string} assetUid - System asset UID
 * @returns {Promise<number>} Next available instance_index
 */
export async function getNextInstanceIndex(assetUid) {
  const supabase = await getSupabaseClient();
  const requestLogger = logger.createRequestLogger();

  try {
    const { data, error } = await supabase
      .from('instances')
      .select('instance_index')
      .eq('asset_uid', assetUid)
      .order('instance_index', { ascending: false })
      .limit(1);

    if (error) throw error;

    if (!data || data.length === 0 || data[0].instance_index === null) {
      return 1;
    }

    return data[0].instance_index + 1;

  } catch (error) {
    requestLogger.error('Repository error getting next instance index', {
      error: error.message,
      assetUid
    });
    throw error;
  }
}
